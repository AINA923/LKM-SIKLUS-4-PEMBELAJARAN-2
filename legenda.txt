/*
  DOCX mandiri tanpa pustaka eksternal.
  DOCX adalah paket ZIP berisi XML. ZIP di bawah memakai metode STORE
  (tanpa kompresi), dengan CRC32 dan nama berkas ASCII.
*/
const encoder = new TextEncoder();

function xml(text) {
  return String(text).replace(/[<>&"']/g, char => ({
    "<":"&lt;",">":"&gt;","&":"&amp;",'"':"&quot;","'":"&apos;"
  }[char])).replace(
    /[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, ""
  );
}
function paragraph(text, bold=false) {
  return '<w:p><w:pPr><w:spacing w:after="140"/></w:pPr><w:r>'+
    (bold ? '<w:rPr><w:b/></w:rPr>' : '')+
    '<w:t xml:space="preserve">'+xml(text)+'</w:t></w:r></w:p>';
}
function answerParagraphs(text) {
  return text.split(/\r?\n/).map(line => paragraph(line || " ")).join("");
}
const crcTable = new Uint32Array(256);
for (let n=0;n<256;n++) {
  let c=n;
  for (let k=0;k<8;k++) c=(c&1) ? 0xedb88320^(c>>>1) : c>>>1;
  crcTable[n]=c>>>0;
}
function crc32(bytes) {
  let crc=0xffffffff;
  for (const byte of bytes) crc=crcTable[(crc^byte)&255]^(crc>>>8);
  return (crc^0xffffffff)>>>0;
}
function zip(entries) {
  const localParts=[], centralParts=[];
  let offset=0, centralSize=0;

  for (const [path,content] of entries) {
    const name=encoder.encode(path), bytes=encoder.encode(content);
    const crc=crc32(bytes);
    const local=new Uint8Array(30+name.length);
    const lv=new DataView(local.buffer);
    lv.setUint32(0,0x04034b50,true);
    lv.setUint16(4,20,true);
    lv.setUint16(6,0,true);
    lv.setUint16(8,0,true);
    lv.setUint16(12,33,true);
    lv.setUint32(14,crc,true);
    lv.setUint32(18,bytes.length,true);
    lv.setUint32(22,bytes.length,true);
    lv.setUint16(26,name.length,true);
    local.set(name,30);

    const central=new Uint8Array(46+name.length);
    const cv=new DataView(central.buffer);
    cv.setUint32(0,0x02014b50,true);
    cv.setUint16(4,20,true);
    cv.setUint16(6,20,true);
    cv.setUint16(14,33,true);
    cv.setUint32(16,crc,true);
    cv.setUint32(20,bytes.length,true);
    cv.setUint32(24,bytes.length,true);
    cv.setUint16(28,name.length,true);
    cv.setUint32(42,offset,true);
    central.set(name,46);

    localParts.push(local,bytes);
    centralParts.push(central);
    offset+=local.length+bytes.length;
    centralSize+=central.length;
  }
  const end=new Uint8Array(22);
  const ev=new DataView(end.buffer);
  ev.setUint32(0,0x06054b50,true);
  ev.setUint16(8,entries.length,true);
  ev.setUint16(10,entries.length,true);
  ev.setUint32(12,centralSize,true);
  ev.setUint32(16,offset,true);

  return new Blob([...localParts,...centralParts,end], {
    type:"application/vnd.openxmlformats-officedocument.wordprocessingml.document"
  });
}
function makeDocx(d) {
  let body=paragraph("LEMBAR KERJA MURID",true)+
    paragraph("Bahasa Indonesia — Cerita Legenda",true)+
    paragraph("Guru Praktik: Aina Wirda, S.Pd")+
    paragraph("Nama: "+d.nama)+paragraph("Kelas: "+d.kelas)+
    paragraph("Tanggal: "+d.tanggal)+
    paragraph("Judul cerita: "+d.judulcerita);

  questions.forEach(([title,instruction,ids]) => {
    body+=paragraph(title,true)+paragraph(instruction);
    ids.forEach((id,i) => {
      if (ids.length>1) body+=paragraph("Pertanyaan "+(i+1)+":",true);
      body+=answerParagraphs(d[id]);
    });
  });

  const declaration='<?xml version="1.0" encoding="UTF-8" standalone="yes"?>';
  const documentXml=declaration+
    '<w:document xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main">'+
    '<w:body>'+body+
    '<w:sectPr><w:pgSz w:w="11906" w:h="16838"/>'+
    '<w:pgMar w:top="1134" w:right="1134" w:bottom="1134" w:left="1134"/>'+
    '</w:sectPr></w:body></w:document>';

  const types=declaration+
    '<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">'+
    '<Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/>'+
    '<Default Extension="xml" ContentType="application/xml"/>'+
    '<Override PartName="/word/document.xml" '+
    'ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.document.main+xml"/>'+
    '</Types>';

  const rels=declaration+
    '<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">'+
    '<Relationship Id="rId1" '+
    'Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" '+
    'Target="word/document.xml"/></Relationships>';

  return zip([
    ["[Content_Types].xml",types],
    ["_rels/.rels",rels],
    ["word/document.xml",documentXml]
  ]);
}
function filename(d) {
  const clean = value => value.replace(/[^a-zA-Z0-9_-]/g,"_").slice(0,60);
  return "LKM_Legenda_"+clean(d.nama)+"_"+clean(d.kelas)+".docx";
}
